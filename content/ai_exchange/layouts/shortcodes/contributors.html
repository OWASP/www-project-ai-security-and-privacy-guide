{{- /* Contributors hall of fame, rendered as paged cards. Data comes from content/contributors.csv (semicolon-separated): Name ; Organization ; Country ; Contributions ; Lead Line 1 is the header and is skipped. Empty or "-" means unknown. Put "yes" in the Lead column to show the "Leadership team" badge. All cards are rendered at build time; the script at the bottom only adds paging and search, so without JS everything still shows as a plain grid. Avatar colours are picked from a hash of the name so they don't shuffle when someone is added. */ -}} {{- $sep := ";" -}} {{- $raw := os.ReadFile "content/contributors.csv" -}} {{- $raw = replace $raw "\r" "" -}} {{- $lines := split (trim $raw "\n") "\n" -}} {{- $people := slice -}} {{- range after 1 $lines }} {{- $line := . -}} {{- if trim $line " \t" }} {{- $c := split $line $sep -}} {{- if lt (len $c) 4 }}{{ range seq (sub 4 (len $c)) }}{{ $c = $c | append "" }}{{ end }}{{ end -}} {{- $lead := "" }}{{ if gt (len $c) 4 }}{{ $lead = trim (index $c 4) " " }}{{ end -}} {{- $org := trim (index $c 1) " " }}{{ if eq $org "-" }}{{ $org = "" }}{{ end -}} {{- $country := trim (index $c 2) " " }}{{ if eq $country "-" }}{{ $country = "" }}{{ end -}} {{- $people = $people | append (dict "name" (trim (index $c 0) " ") "organization" $org "country" $country "contrib" (trim (index $c 3) " ") "lead" (ne $lead "")) -}} {{- end }} {{- end -}} {{- $countries := slice -}} {{- /* Count countries once, e.g. "London, United Kingdom" and "UK" are the same */ -}} {{- $aliases := dict "united kingdom" "uk" "great britain" "uk" "england" "uk" "united states" "us" "usa" "us" "united states of america" "us" -}} {{- range $people }} {{- if .country }} {{- $parts := split .country "," -}} {{- $k := lower (trim (index $parts (sub (len $parts) 1)) " ") -}} {{- $k = index $aliases $k | default $k -}} {{- $countries = $countries | append $k -}} {{- end }} {{- end -}} {{- $palette := slice "#2E7D32" "#2F6F86" "#1F2937" "#8A4B08" "#5B3F8C" "#9F1D4F" -}} {{- $tints := slice "#DEEADE" "#DEE8EC" "#DBDDDF" "#ECE2D7" "#E5E0ED" "#F0DBE3" -}}
