{{- /* Contributors hall of fame — reads a simple semicolon-separated file and renders BOTH the desktop table and the mobile cards from the same data (at build time). Edit content/contributors.csv to add or change contributors: * One contributor per line. * Columns separated by a semicolon: Name ; Organization ; Country ; Contributions ; Lead * No quotes, no field names to repeat. Spaces around the ";" don't matter (trimmed). * Leave a column empty if unknown (trailing empty columns may be omitted). * Put "yes" in the 5th "Lead" column to show a leadership name in bold. * Line 1 is the header row and is skipped — keep it. To use a different separator (e.g. a pipe), change $sep below to "|". Mobile note: the cards below are rendered from the data directly, so they never depend on JavaScript reading the (hidden) table. The small script only powers the optional "Show More" collapse; without it, all cards simply show. */ -}} {{- $sep := ";" -}} {{- $raw := os.ReadFile "content/contributors.csv" -}} {{- $raw = replace $raw "\r" "" -}} {{- $lines := split (trim $raw "\n") "\n" -}} {{- $people := slice -}} {{- range after 1 $lines }} {{- $line := . -}} {{- if trim $line " \t" }} {{- $c := split $line $sep -}} {{- if lt (len $c) 4 }}{{ range seq (sub 4 (len $c)) }}{{ $c = $c | append "" }}{{ end }}{{ end -}} {{- $lead := "" }}{{ if gt (len $c) 4 }}{{ $lead = trim (index $c 4) " " }}{{ end -}} {{- $people = $people | append (dict "name" (trim (index $c 0) " ") "organization" (trim (index $c 1) " ") "country" (trim (index $c 2) " ") "contrib" (trim (index $c 3) " ") "lead" (ne $lead "")) -}} {{- end }} {{- end -}}
